cmake_minimum_required(VERSION 3.8)
project(luna_mechanism)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/mechanism_stub_gateway scripts/mechanism_demo
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config urdf DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(launch_testing_ament_cmake REQUIRED)
  ament_add_pytest_test(policy test/test_policy.py)
  add_launch_test(test/test_stub_launch.py TIMEOUT 90)
  find_package(ament_lint_auto REQUIRED)
  list(APPEND AMENT_LINT_AUTO_EXCLUDE ament_cmake_copyright ament_cmake_cpplint)
  ament_lint_auto_find_test_dependencies()
  # This package has no C++ files; cpplint would fail with "No files found".
endif()
ament_package()
